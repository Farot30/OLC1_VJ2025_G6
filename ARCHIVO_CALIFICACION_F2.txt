int a;

PROC actualizar_a(int: b, bool: mult){
	if (mult) {
		a = a * b;
	} else {
		a = a + b;
	}
}

EXEC actualizar_a(1,False);
println(a);

EXEC actualizar_a(2,True);
println(a);


PROC fibonacci(int: numero){
	int a;
	int b;
	int temp;
	int i;
    	if (numero == 0) {
        	a = 0;
        	println(a);
    	} else if (numero == 1) {
        	a = 1;
        	println(a);
    	} else {
        	a = 0;
        	b = 1;
        	i = 2;

        	while (i <= numero) {
            	temp = a + b;
            	a = b;
            	b = temp;
            	i = i + 1;
        	}

        	println(b);
    	}
}

EXEC Fibonacci(0); // Res: 0
EXEC Fibonacci(1); // Res: 1
EXEC Fibonacci(7); // Res: 13
