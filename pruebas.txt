// 1. SWITCH ANIDADOS CON IF - Con errores
int opcion1 = 2;
int opcion2 = 1;

println("--- PRUEBA 1: SWITCH ANIDADOS CON IF (CON ERRORES) ---");
switch(opcion1) {
    case 1:
        println("Opcion1 es 1");
        break;
    // ERROR: Caso duplicado
    case 1:
        println("Opcion1 es 1 de nuevo");
        break;
    case 2:
        println("Opcion1 es 2");
        
        // Switch anidado con error
        switch(opcion2 + true) {  // ERROR: Suma incompatible de tipos
            case 1:
                println("Opcion2 es 1");
                
                // If anidado con error
                if(opcion1 + "error") {  // ERROR: Concatenación inválida
                    println("Opcion1 es mayor que Opcion2");
                } else {
                    println("Opcion1 NO es mayor que Opcion2");
                }
                break;
                
            case 2:
                println("Opcion2 es 2");
                break;
        }
        break;
        
    default:
        println("Opcion1 no es 1 ni 2");
        // ERROR: Continue fuera de un ciclo
        continue;
}

// 4. COMBINACIÓN DE ESTRUCTURAS - Con errores
println("\n--- PRUEBA 4: COMBINACIÓN DE ESTRUCTURAS (CON ERRORES) ---");
int x = 1;

switch(x) {
    case "1":  // ERROR: Tipo incorrecto para case (string vs int)
        println("x es 1");
        
        // For dentro de switch con error
        for(int i = "cero"; i < 2; i++) {  // ERROR: Inicialización con tipo incorrecto
            println("  FOR dentro de SWITCH: i = " + i);
            
            // ERROR: Operación inválida
            i = i / 0;
            
            if(i == 0) {
                // ERROR: Break anidado excesivo
                break;
            }
        }
        break;
}

        case 1:
