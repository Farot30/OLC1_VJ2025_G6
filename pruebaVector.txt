// Pruebas de funciones nativas básicas
println("Funciones nativas con números:");
println(seno(0.5));
println(coseno(0.5));
println(inv(4));

// Con variables
float angulo = 3.14159 / 2;  // 90 grados en radianes
println("seno(π/2) = " + seno(angulo));  // Debería ser cercano a 1
println("coseno(π/2) = " + coseno(angulo));  // Debería ser cercano a 0
println("inv(10) = " + inv(10));  // Debería ser 0.1


// Vector de enteros
vector[int] numeros (5) = [10, 20, 30, 40, 50];
println("Vector original: " + numeros);

// Modificar elementos
numeros[2] = 300;
println("Vector modificado: " + numeros);

// Funciones con elementos de vector
println("seno del primer elemento: " + seno(300));
println("inverso del último elemento: " + inv(40));

// Shuffle del vector completo
println("Vector aleatorizado: " + shuffle(numeros));

// Matriz 3x3
vector[int] matriz (3, 3) = [[1, 2, 3], [4, 5, 6], [7, 8, 9]];
println("Matriz original:");
println(matriz);

// Modificar elementos específicos
matriz[0][0] = 100;
matriz[1][1] = 500;
matriz[2][2] = 900;
println("Matriz modificada:");
println(matriz);

// Vector 3D (cubo 2x2x2)
vector[int] cubo (2, 2, 2) = [[[1, 2], [3, 4]], [[5, 6], [7, 8]]];
println("Cubo 3D:");
println(cubo);
cubo[1][0][1] = 60;  // Modificar un elemento interno
println("Cubo modificado:");
println(cubo);

// Vector de flotantes
vector[float] decimales (3) = [1.5, 2.7, 3.9];
println("Vector de flotantes: " + decimales);

// Vector de booleanos
vector[bool] banderas (4) = [true, false, true, true];
println("Vector de booleanos: " + banderas);

// Vector de cadenas
vector[str] palabras (3) = ["Hola", "Mundo", "Vectorial"];
println("Vector de cadenas: " + palabras);
